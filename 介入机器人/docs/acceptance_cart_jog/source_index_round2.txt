7: struct ArmCartJogReference {
8:     static constexpr double PI = 3.14159265358979323846;
9:     static constexpr double RAD = PI / 180.0;
10:     static constexpr double L1 = 200.0, L2 = 180.0;
11:     static constexpr double T_ACC = 1.0, T_DEC = 0.5;
12:     static constexpr int INITIAL_BRANCH = 1, BISECTION_COUNT = 12;
13:     static constexpr std::array<double, 5> JOINT_VCAP{{24,10,10,10,10}};
14:     std::array<double, 5> p{}, v{}, a{}, anchor{};
15:     double path_speed = 0, target_speed = 0, path_v = 0;
16:     double v0 = 0, v1 = 0, duration = T_ACC, tau = 0, sigma = 0;

18:     int latched_mode = 0, latched_dir = 0, branch = INITIAL_BRANCH;
19:     bool limited = false;
20: 
21:     void step(bool start, bool held, int mode, int dir, double speed,
22:         const std::array<double, 5>& seed, double dt)
23:     {
24:         bool reanchor = start;
25:         if (start) {
26:             p = seed;
27:             for (int i = 0; i < 5; ++i) { v[i] = 0; a[i] = 0; }
28:             path_v = 0; v0 = 0; v1 = 0; tau = 0; duration = T_ACC;
29:             latched_mode = mode; latched_dir = dir;
30:         }
31:         const auto position_before = p, velocity_before = v;
32:         const double path_v_previous = path_v;
33:         target_speed = 0;
34:         if (held && mode == latched_mode && dir == latched_dir) target_speed = speed;
35:         if (target_speed != v1) {
36:             v0 = path_v; v1 = target_speed;
37:             if (v1 > v0) duration = T_ACC; else duration = T_DEC;
38:             tau = 0;
39:         }
40:         if (!start) tau += dt;
41:         const double u = std::min(1.0, tau / duration);
42:         path_v = v0 + (v1-v0)*(3.0*u*u-2.0*u*u*u);
43:         if (path_v == 0 && v1 == 0 && held
44:             && (mode != latched_mode || dir != latched_dir)) reanchor = true;
45:         if (reanchor) {
46:             anchor = p; latched_mode = mode; latched_dir = dir; sigma = 0;
47:             double branch_q3 = anchor[2];
48:             while (branch_q3 > 180.0) branch_q3 -= 360.0;
49:             while (branch_q3 < -180.0) branch_q3 += 360.0;
50:             if (branch_q3 != 0.0 && branch_q3 != 180.0 && branch_q3 != -180.0) {
51:                 if (branch_q3 < 0) branch = 1; else branch = -1;
52:             }
53:             wx = L1*std::sin(anchor[1]*RAD)+L2*std::sin((anchor[1]+anchor[2])*RAD+PI);
54:             wy = -L1*std::cos(anchor[1]*RAD)-L2*std::cos((anchor[1]+anchor[2])*RAD+PI);
55:         }
56:         limited = false;
57:         if (start) { path_speed = 0; target_speed = 0; return; }
58:         const double delta_sigma_des = 0.5*(path_v_previous+path_v)*dt;
59:         double delta_sigma;
60:         if (latched_mode >= 3) {
61:             const int j = latched_mode == 3 ? 0 : latched_mode == 4 ? 4 : 3;
62:             delta_sigma = std::min(delta_sigma_des, JOINT_VCAP[j]*dt);
63:             const double s = latched_dir*(sigma+delta_sigma);
64:             p = anchor;
65:             const double axis_sign = latched_mode == 5 ? 1.0 : -1.0;
66:             p[j] = anchor[j]+axis_sign*s;
67:         } else {
68:             const double ex = latched_mode == 1 ? 1.0 : 0.0;
69:             const double ey = latched_mode == 1 ? 0.0 : 1.0;
70:             double best_step = 0, lower_step = 0, upper_step = delta_sigma_des;
71:             double best_wx = 0, best_wy = 0, best_q2 = 0, best_q3 = 0, best_q4 = 0;
72:             for (int i = 0; i <= BISECTION_COUNT; ++i) {
73:                 const double trial_step = i == 0 ? delta_sigma_des : (lower_step+upper_step)/2.0;
74:                 const double candidate_wx = wx+latched_dir*trial_step*ex;
75:                 const double candidate_wy = wy+latched_dir*trial_step*ey;
76:                 const double candidate_radius = std::sqrt(candidate_wx*candidate_wx+candidate_wy*candidate_wy);
77:                 bool trial_ok = false;
78:                 double candidate_q2 = 0, candidate_q3 = 0, candidate_q4 = 0;
79:                 if (candidate_radius >= L1-L2 && candidate_radius <= L1+L2) {
80:                     double candidate_cosine = (candidate_radius*candidate_radius-L1*L1-L2*L2)/(2.0*L1*L2);
81:                     if (candidate_cosine > 1.0) candidate_cosine = 1.0;
82:                     else if (candidate_cosine < -1.0) candidate_cosine = -1.0;
83:                     const double candidate_elbow = branch*std::acos(candidate_cosine);
84:                     const double azimuth_denominator = -candidate_wy;
85:                     double candidate_azimuth;
86:                     if (azimuth_denominator > 0) candidate_azimuth = std::atan(candidate_wx/azimuth_denominator);
87:                     else if (azimuth_denominator < 0) {
88:                         if (candidate_wx >= 0) candidate_azimuth = std::atan(candidate_wx/azimuth_denominator)+PI;
89:                         else candidate_azimuth = std::atan(candidate_wx/azimuth_denominator)-PI;
90:                     } else {
91:                         if (candidate_wx > 0) candidate_azimuth = PI/2.0;
92:                         else candidate_azimuth = -PI/2.0;
93:                     }
94:                     candidate_q2 = (candidate_azimuth-std::atan(L2*std::sin(candidate_elbow)/(L1+L2*std::cos(candidate_elbow))))/RAD;
95:                     candidate_q3 = (candidate_elbow-PI)/RAD;
96:                     while (candidate_q2-p[1] > 180.0) candidate_q2 -= 360.0;
97:                     while (candidate_q2-p[1] < -180.0) candidate_q2 += 360.0;
98:                     while (candidate_q3-p[2] > 180.0) candidate_q3 -= 360.0;
99:                     while (candidate_q3-p[2] < -180.0) candidate_q3 += 360.0;
100:                     candidate_q4 = anchor[3]+(anchor[1]+anchor[2])-(candidate_q2+candidate_q3);
101:                     trial_ok = std::abs(candidate_q2-p[1]) <= JOINT_VCAP[1]*dt
102:                         && std::abs(candidate_q3-p[2]) <= JOINT_VCAP[2]*dt
103:                         && std::abs(candidate_q4-p[3]) <= JOINT_VCAP[3]*dt;
104:                 }
105:                 if (trial_ok) {
106:                     lower_step = trial_step; best_step = trial_step;
107:                     best_wx = candidate_wx; best_wy = candidate_wy;
108:                     best_q2 = candidate_q2; best_q3 = candidate_q3; best_q4 = candidate_q4;
109:                     if (i == 0) break;
110:                 } else {
111:                     upper_step = trial_step;
112:                 }
113:             }
114:             delta_sigma = best_step;
115:             if (delta_sigma > 0) {
116:                 wx = best_wx; wy = best_wy;
117:                 p[0] = anchor[0]; p[4] = anchor[4];
118:                 p[1] = best_q2; p[2] = best_q3; p[3] = best_q4;
119:             }
120:         }

122:         path_speed = delta_sigma/dt;

125:             v[i] = (p[i]-position_before[i])/dt;
126:             a[i] = (v[i]-velocity_before[i])/dt;
127:         }
128:     }

41:     require(config.base_offset_mm.allFinite(), "基座平移偏移必须为有限毫米值");
42:     for (int i = 0; i < AxisCount; ++i) {
43:         require(!std::isnan(config.lower[i]) && !std::isnan(config.upper[i])
44:             && config.lower[i] < config.upper[i],
45:             "轴限位必须有序；使用正负无穷表示不设软件限位");
46:     }
47:     require(config.max_velocity.allFinite()
48:         && (config.max_velocity.array() > 0).all(),
49:         "五轴速度限位必须为正");
50:     for (const auto& calibration : config.calibration) {
51:         require(std::isfinite(calibration.zero) && positive(calibration.scale)
52:             && (calibration.direction == -1 || calibration.direction == 1),
53:             "编码器标定参数无效");
54:     }
55:     for (double value : {options.position_tolerance,
56:         options.singularity_stop, options.singularity_slow,
57:         options.max_translation, options.max_rotation}) {
58:         require(positive(value), "求解器参数必须为正");
59:     }
60:     require(options.singularity_stop < options.singularity_slow
61:         && options.max_rotation < Pi,
62:         "奇异阈值或角度步长无效");

78: ArmPose PositioningArmKinematics::forward(const JointVector& joints) const
79: {
80:     require(checkLimits(joints), "轴位置无效或超出限位");
81: 
82:     ArmPose result;
83:     // 基座平移只改变原点；轴1的Z符号受归零逐位回归约束，见交付报告。
84:     result.position << 0, 0, joints[0];
85: 
86:     const double first_link_yaw = joints[1];
87:     result.position += config_.link_lengths[0]
88:         * Eigen::Vector3d(std::sin(first_link_yaw), -std::cos(first_link_yaw), 0);
89: 
90:     // 轴3机械零位使第二根连杆反向；工具偏置仍按轴2+轴3+轴4定义。
91:     const double second_link_yaw = first_link_yaw + joints[2] + Pi;
92:     result.position += config_.link_lengths[1]
93:         * Eigen::Vector3d(std::sin(second_link_yaw), -std::cos(second_link_yaw), 0);
94: 
95:     const double tool_yaw = joints[1] + joints[2] + joints[3] - Pi / 2;
96:     const Eigen::Quaterniond wrist_rotation = rotationZ(tool_yaw)
97:         * Eigen::Quaterniond(Eigen::AngleAxisd(joints[4], Eigen::Vector3d::UnitX()));
98:     result.position += wrist_rotation
99:         * Eigen::Vector3d(config_.lateral_offset, -config_.tool_length, 0);
100:     result.position += config_.base_offset_mm * 0.001;
101:     result.quaternion = normalized(wrist_rotation);
102:     if (result.quaternion.w() < 0) result.quaternion.coeffs() *= -1;
103:     return result;
104: }

106: FrameJacobian PositioningArmKinematics::computeFrameJacobian(
107:     const JointVector& joints) const
108: {
109:     require(checkLimits(joints), "轴位置无效或超出限位");
110: 
111:     const ArmPose pose = forward(joints);
112:     FrameJacobian result = FrameJacobian::Zero();
113:     result.block<3, 1>(0, 0) = Eigen::Vector3d::UnitZ();
114: 
115:     const double z = joints[0];
116:     Eigen::Vector3d joint_origin = config_.base_offset_mm * 0.001;
117:     joint_origin.z() += z;
118:     double yaw = 0;
119:     for (int i = 0; i < 3; ++i) {
120:         const Eigen::Vector3d axis = Eigen::Vector3d::UnitZ();
121:         result.block<3, 1>(0, i + 1) = axis.cross(pose.position - joint_origin);
122:         result.block<3, 1>(3, i + 1) = axis;
123:         yaw += joints[i + 1];
124:         if (i == 0) {
125:             joint_origin += config_.link_lengths[0]
126:                 * Eigen::Vector3d(std::sin(yaw), -std::cos(yaw), 0);
127:         } else if (i == 1) {
128:             joint_origin += config_.link_lengths[1]
129:                 * Eigen::Vector3d(std::sin(yaw + Pi), -std::cos(yaw + Pi), 0);
130:         }
131:     }
132: 
133:     const Eigen::Vector3d pitch_axis_world =
134:         rotationZ(yaw - Pi / 2) * Eigen::Vector3d::UnitX();
135:     result.block<3, 1>(0, 4) =
136:         pitch_axis_world.cross(pose.position - joint_origin);
137:     result.block<3, 1>(3, 4) = pitch_axis_world;
138:     return result;
139: }

148: bool PositioningArmKinematics::isNearSingularity(const JointVector& joints) const
149: {
150:     require(checkLimits(joints), "轴位置无效或超出限位");
151: 
152:     // 平面机构在轴3伸直或折叠时，优先按机械结构直接判定。
153:     if (std::abs(std::sin(joints[2])) < options_.singularity_slow) return true;
154: 
155:     const auto full = computeFrameJacobian(joints);
156:     Eigen::Matrix3d jacobian;
157:     jacobian.topRows<2>() = full.block<2, 3>(0, 1) / characteristic_length_;
158:     jacobian.bottomRows<1>().setOnes();
159:     return Eigen::JacobiSVD<Eigen::Matrix3d>(jacobian)
160:         .singularValues().minCoeff() < options_.singularity_slow;

197: IkResult PositioningArmKinematics::solveCartesianJog(
198:     const JointVector& current, const CartesianJog& command) const
199: {
200:     IkResult result;
201:     result.target = current;
202:     if (!current.allFinite() || !command.translation.allFinite()
203:         || !std::isfinite(command.yaw) || !std::isfinite(command.pitch)
204:         || !positive(command.duration) || command.duration > 1) {
205:         return result;
206:     }
207:     if (!checkLimits(current)) {
208:         result.status = IkStatus::JointLimit;
209:         return result;
210:     }
211:     const ArmPose current_pose = forward(current);
212:     result.pose = current_pose;
213:     const auto active = (command.translation.array() != 0).count()
214:         + (command.yaw != 0) + (command.pitch != 0);
215:     if (active > 1) return result;
216:     if (command.translation.norm() > options_.max_translation
217:         || std::abs(command.yaw) + std::abs(command.pitch) > options_.max_rotation) {
218:         result.status = IkStatus::StepLimit;
219:         return result;
220:     }
221: 
222:     // 升降、抬头和静止命令不经过平面逆解，也不受平面机械奇异影响。
223:     if (active == 0 || command.translation.z() != 0 || command.pitch != 0) {
224:         JointVector next = current;
225:         next[0] += command.translation.z();
226:         next[4] -= command.pitch;
227:         if (!checkLimits(next)) {
228:             result.status = IkStatus::JointLimit;
229:             return result;
230:         }
231:         if (((next - current).cwiseAbs().array()
232:             > (config_.max_velocity * command.duration).array()).any()) {
233:             result.status = IkStatus::StepLimit;
234:             return result;
235:         }
236:         result.status = IkStatus::Success;
237:         result.target = next;
238:         result.pose = forward(next);
239:         result.pose.quaternion = continuousQuaternion(result.pose.quaternion, current_pose.quaternion);
240:         result.position_error = result.orientation_error = 0;
241:         return result;
242:     }
243: 
244:     ArmPose target = current_pose;
245:     target.position += command.translation;
246:     target.quaternion = normalized(
247:         rotationZ(command.yaw) * current_pose.quaternion);
248:     target.quaternion = continuousQuaternion(target.quaternion, current_pose.quaternion);
249: 
250:     const JointVector travel = config_.max_velocity * command.duration;
251:     if (!travel.allFinite()) return result;
252: 
253:     const auto planar_sigma = [&](const JointVector& joints) {
254:         const auto full = computeFrameJacobian(joints);
255:         Eigen::Matrix3d planar;
256:         planar.topRows<2>() = full.block<2, 3>(0, 1) / characteristic_length_;
257:         planar.bottomRows<1>().setOnes();
258:         return (std::min)(std::abs(std::sin(joints[2])),
259:             Eigen::JacobiSVD<Eigen::Matrix3d>(planar).singularValues().minCoeff());
260:     };
261:     if (planar_sigma(current) < options_.singularity_stop) {
262:         result.status = IkStatus::NearSingularity;
263:         return result;
264:     }
265: 
266:     // 解析解只处理平面三轴：先扣除工具偏置，再解轴2、3两连杆，最后由总偏航角解轴4。
267:     const double target_yaw = current[1] + current[2] + current[3]
268:         - Pi / 2 + command.yaw;
269:     const double pitch = current[4];
270:     const double tool_x = config_.lateral_offset * std::cos(target_yaw)
271:         + config_.tool_length * std::cos(pitch) * std::sin(target_yaw);
272:     const double tool_y = config_.lateral_offset * std::sin(target_yaw)
273:         - config_.tool_length * std::cos(pitch) * std::cos(target_yaw);
274:     const double wrist_x = target.position.x() - config_.base_offset_mm.x() * 0.001 - tool_x;
275:     const double wrist_y = target.position.y() - config_.base_offset_mm.y() * 0.001 - tool_y;
276:     const double radius = std::hypot(wrist_x, wrist_y);
277:     const double minimum_radius = std::abs(config_.link_lengths[0]
278:         - config_.link_lengths[1]);
279:     const double maximum_radius = config_.link_lengths.sum();
280:     const double tolerance = (std::max)(options_.position_tolerance, 1e-12);
281:     if (!std::isfinite(radius)
282:         || radius < minimum_radius - tolerance
283:         || radius > maximum_radius + tolerance) {
284:         result.status = IkStatus::NoConvergence;
285:         return result;
286:     }
287: 
288:     const double cosine_q3 = (std::max)(-1.0, (std::min)(1.0,
289:         (radius * radius - config_.link_lengths[0] * config_.link_lengths[0]
290:             - config_.link_lengths[1] * config_.link_lengths[1])
291:         / (2 * config_.link_lengths[0] * config_.link_lengths[1])));
292:     const double elbow = std::acos(cosine_q3);
293:     const double current_branch = std::sin(current[2]);
294:     if (std::abs(current_branch) < options_.singularity_stop) {
295:         result.status = IkStatus::NearSingularity;
296:         return result;
297:     }
298: 
299:     const auto nearest_equivalent = [](double principal, double reference) {
300:         return reference + std::remainder(
301:             principal - std::remainder(reference, 2 * Pi), 2 * Pi);
302:     };
303:     JointVector best = current;
304:     double best_cost = std::numeric_limits<double>::infinity();
305:     IkStatus blocked = IkStatus::NoConvergence;
306:     for (double branch : {1.0, -1.0}) {
307:         const double effective_q3 = branch * elbow;
308:         const double q3 = effective_q3 - Pi;
309:         const double q2 = std::atan2(wrist_x, -wrist_y)
310:             - std::atan2(config_.link_lengths[1] * std::sin(effective_q3),
311:                 config_.link_lengths[0] + config_.link_lengths[1] * std::cos(effective_q3));
312:         JointVector candidate = current;
313:         candidate[1] = nearest_equivalent(q2, current[1]);
314:         candidate[2] = nearest_equivalent(q3, current[2]);
315:         candidate[3] = nearest_equivalent(
316:             target_yaw + Pi / 2 - candidate[1] - candidate[2], current[3]);
317: 
318:         if (current_branch * std::sin(candidate[2]) <= 0) {
319:             blocked = IkStatus::NearSingularity;
320:             continue;
321:         }
322:         if (!checkLimits(candidate)) {
323:             blocked = IkStatus::JointLimit;
324:             continue;
325:         }
326:         if (((candidate - current).cwiseAbs().array() > travel.array()).any()) {
327:             blocked = IkStatus::StepLimit;
328:             continue;
329:         }
330:         if (planar_sigma(candidate) < options_.singularity_stop) {
331:             blocked = IkStatus::NearSingularity;
332:             continue;
333:         }
334: 
335:         const double cost = (candidate.tail<4>() - current.tail<4>()).squaredNorm();
336:         if (cost < best_cost) {
337:             best = candidate;
338:             best_cost = cost;
339:         }
340:     }
341:     if (!std::isfinite(best_cost)) {
342:         result.status = blocked;
343:         return result;
344:     }
345: 
346:     // 解析求解成功后直接返回请求姿态；不再用一次正运动学迭代复核。
347:     result.status = IkStatus::Success;
348:     result.target = best;
349:     result.pose = target;
350:     result.position_error = result.orientation_error = 0;
351:     return result;
352: }

354: IkResult PositioningArmKinematics::solveJog(
355:     const JointVector& current, const JogCommand& command) const
356: {
357:     CartesianJog jog;
358:     jog.duration = command.duration;
359:     double scale = 1;
360:     if (checkLimits(current) && (command.axis == JogAxis::X
361:         || command.axis == JogAxis::Y || command.axis == JogAxis::Yaw)) {
362:         const auto full = computeFrameJacobian(current);
363:         Eigen::Matrix3d planar;
364:         planar.topRows<2>() = full.block<2, 3>(0, 1) / characteristic_length_;
365:         planar.bottomRows<1>().setOnes();
366:         const double sigma = (std::min)(std::abs(std::sin(current[2])),
367:             Eigen::JacobiSVD<Eigen::Matrix3d>(planar).singularValues().minCoeff());
368:         // 硬停止区保留非零命令，让逆解明确报告奇异，而非伪装成静止成功。
369:         if (sigma >= options_.singularity_stop)
370:             scale = std::min(1.0, sigma / options_.singularity_slow);
371:     }
372:     const double displacement = command.speed * command.duration * scale;
373:     switch (command.axis) {
374:     case JogAxis::X: jog.translation.x() = displacement; break;
375:     case JogAxis::Y: jog.translation.y() = displacement; break;
376:     case JogAxis::Z: jog.translation.z() = displacement; break;
377:     case JogAxis::Pitch: jog.pitch = displacement; break;
378:     case JogAxis::Yaw: jog.yaw = displacement; break;
379:     default: {
380:         IkResult result;
381:         result.target = current;
382:         return result;
383:     }
384:     }
385:     auto result = solveCartesianJog(current, jog);
386:     result.speed_scale = scale;
387:     return result;
388: }
